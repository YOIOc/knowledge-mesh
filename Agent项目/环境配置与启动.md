## MySQL
```
# 下载镜像
docker pull mysql:5.7

# 创建数据卷
docker volume create mysql57-data

# 启动容器
docker run -d --name mysql57 -p 3306:3306 --restart unless-stopped -e MYSQL_ROOT_PASSWORD=12345678 -e MYSQL_ROOT_HOST=% -e TZ=Asia/Shanghai -v mysql57-data:/var/lib/mysql mysql:5.7 --character-set-server=utf8mb4 --collation-server=utf8mb4_unicode_ci
```
## PostgresSql
```
# 下载镜像
docker pull pgvector/pgvector:pg17

# 创建数据卷
docker volume create pgvector-data

# 启动容器
docker run -d --name pgvector -p 5432:5432 --restart unless-stopped -e POSTGRES_USER=postgres -e POSTGRES_PASSWORD=12345678 -e TZ=Asia/Shanghai -v pgvector-data:/var/lib/postgresql/data pgvector/pgvector:pg17

# 初始化 PGVector 扩展
docker exec -it pgvector psql -U postgres -d postgres

# 检查 PGVector 是否安装成功
CREATE EXTENSION IF NOT EXISTS vector;
SELECT extname, extversion FROM pg_extension WHERE extname = 'vector';
```
## Redis
```
# 下载镜像
docker pull redis:6.0.8

# 启动容器
docker run -d --name redis -p 0.0.0.0:6379:6379 --restart unless-stopped redis:6.0.8 redis-server --requirepass 12345678 --bind 0.0.0.0
```
## MinIO
```
# 下载镜像
docker pull minio/minio:RELEASE.2024-01-01T16-36-33Z

# 创建数据卷
docker volume create minio-data

# 启动容器
docker run -d --name minio -p 9000:9000 -p 9001:9001 --restart unless-stopped -e MINIO_ROOT_USER=root -e MINIO_ROOT_PASSWORD=12345678 -v minio-data:/data minio/minio:RELEASE.2024-01-01T16-36-33Z server /data --console-address ":9001"
```
