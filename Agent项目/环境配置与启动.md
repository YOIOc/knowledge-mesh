## MySQL
```
# 下载镜像
docker pull mysql:5.7

# 创建数据卷
docker volume create mysql57-data

# 启动容器
docker run -d --name mysql57 -p 3306:3306 --restart unless-stopped -e MYSQL_ROOT_PASSWORD=12345678 -e MYSQL_ROOT_HOST=% -e TZ=Asia/Shanghai -v mysql57-data:/var/lib/mysql mysql:5.7 --character-set-server=utf8mb4 --collation-server=utf8mb4_unicode_ci
```
## PostgresSql
```
# 下载镜像
docker pull pgvector/pgvector:pg17

# 创建数据卷
docker volume create pgvector-data

# 启动容器
docker run -d --name pgvector -p 5432:5432 --restart unless-stopped -e POSTGRES_USER=postgres -e POSTGRES_PASSWORD=12345678 -e TZ=Asia/Shanghai -v pgvector-data:/var/lib/postgresql/data pgvector/pgvector:pg17

# 初始化 PGVector 扩展
docker exec -it pgvector psql -U postgres -d postgres

# 检查 PGVector 是否安装成功(可以查询到 vector 扩展版本)
CREATE EXTENSION IF NOT EXISTS vector;
SELECT extname, extversion FROM pg_extension WHERE extname = 'vector';

# 退出 PostgresSql
exit
```

## Zookeeper
```
# 下载镜像
docker pull zookeeper

# 创建本地目录
mkdir D:\docker\zookeeper\conf
mkdir D:\docker\zookeeper\data
mkdir D:\docker\zookeeper\logs

# 创建配置文件
notepad D:\docker\zookeeper\conf\zoo.cfg

# 配置文件内容
tickTime=2000
initLimit=10
syncLimit=5
dataDir=/data
clientPort=2181
maxClientCnxns=60

# 创建网络
docker network create kafka-net

# 启动容器
docker run -d --name zookeeper --network kafka-net --privileged=true -p 2181:2181 --restart=always -v D:\docker\zookeeper\data:/data -v D:\docker\zookeeper\conf:/conf -v D:\docker\zookeeper\logs:/datalog zookeeper
```

## ElasticSearch
```
# 创建网络
docker network create elk

# 下载镜像
docker pull elasticsearch:8.5.2

# 启动容器
docker run -d --name elasticsearch --network elk -p 9200:9200 -p 9300:9300 -e "discovery.type=single-node" -e "ES_JAVA_OPTS=-Xms768m -Xmx768m" -e "xpack.security.http.ssl.enabled=false" -e "xpack.security.enrollment.enabled=true" elasticsearch:8.5.2

# 进入容器  
docker exec -it elasticsearch /bin/bash

# 执行命令配置密码
./bin/elasticsearch-setup-passwords interactive

# 退出容器
exit

# 检查是否安装成功
curl -u elastic:12345678 http://localhost:9200

# 进入 ES 容器，安装 IK 分词器
docker exec -it elasticsearch /bin/bash

# 使用插件管理器安装 IK 分词器(版本需与 ES 版本一致)
./bin/elasticsearch-plugin install https://get.infini.cloud/elasticsearch/analysis-ik/8.5.2

# 退出容器
exit

# 重启容器使插件生效
docker restart elasticsearch

# 验证安装(能看到 analysis-ik 说明安装成功)
docker exec -it elasticsearch ./bin/elasticsearch-plugin list
```
